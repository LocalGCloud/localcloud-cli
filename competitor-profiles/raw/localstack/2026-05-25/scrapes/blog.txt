      });
      root
        .querySelectorAll('.navbar_dropdown-toggle[aria-expanded="true"]')

          el.setAttribute('aria-expanded', 'false');
        });
    }
    function setMenuOpen(open) {
      if (navRoot) navRoot.classList.toggle('is-menu-open', open);
      if (menuBtn) {
        menuBtn.classList.toggle('w--open', open);
        menuBtn.setAttribute('aria-expanded', String(open));
        menuBtn.setAttribute('aria-label', open ? 'Close menu' : 'Open menu');
      }
      if (!open) closeAllDropdowns();
    }
    if (menuBtn && navRoot) {
      menuBtn.addEventListener('click', function () {
        var next = !navRoot.classList.contains('is-menu-open');
        setMenuOpen(next);
      });
    }

      var list = dd.querySelector('.w-dropdown-list');
      var toggle = dd.querySelector('.navbar_dropdown-toggle');
      if (!list || !toggle) return;
      dd.addEventListener('mouseenter', function () {
        if (!mq.matches) return;
        list.classList.add('w--open');
        toggle.setAttribute('aria-expanded', 'true');
      });
      dd.addEventListener('mouseleave', function () {
        if (!mq.matches) return;
        list.classList.remove('w--open');
        toggle.setAttribute('aria-expanded', 'false');
      });

        if (mq.matches) return;

          .forEach(function (l) {
            if (l !== list) l.classList.remove('w--open');
          });
        list.classList.toggle('w--open', !wasOpen);
        toggle.setAttribute('aria-expanded', String(!wasOpen));
      });
