$(".usp_wrap").each(function () {
  let items = $(this).find(".tab_item");
  let lotties = $(this).find(".tab_lottie").hide();
  let content = $(this).find(".tab_content");
  let prevIndex = -1;
  gsap.defaults({ duration: 0.3, ease: "power3.inOut" });
  function triggerLottie(index) {
    // close state
    if (prevIndex > -1) {
      lotties.eq(prevIndex).hide();
      gsap.to(content.eq(prevIndex), { height: 0 });
    }
    // open state
    lotties.eq(index).show();
    gsap.to(content.eq(index), { height: "auto" });
    lotties.eq(index).click();
    // track previous
    prevIndex = index;
  }
  triggerLottie(0);
  items.each(function (index) {
    let itemIndex = index;
    let link = $(this).find(".tab_trigger");
    link.on("click", function () {
      if (itemIndex !== prevIndex) triggerLottie(itemIndex);
    });
  });
});
